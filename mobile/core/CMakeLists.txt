cmake_minimum_required(VERSION 3.16)
project(stl2stp-core VERSION 0.1.0 LANGUAGES C CXX)

include(CTest)
include(GNUInstallDirs)

option(STL2STP_WITH_OCCT "Build STEP/IGES exchange (OFF is mesh-only, not full validation)" ON)
option(STL2STP_BUILD_CLI "Build the command line executable" ON)
set(OCCT_ROOT "" CACHE PATH "OCCT installation prefix")
set(OCCT_INCLUDE_DIR "" CACHE PATH "Override OCCT headers (include/opencascade, inc or include)")
set(OCCT_LIBRARY_DIR "" CACHE PATH "Override OCCT toolkit directory")
set(OCCT_EXTRA_LIBRARIES "" CACHE STRING "Additional dependencies of a custom static OCCT build")

add_library(stl2stp-core src/mesh.cpp src/simplify.cpp src/weld.cpp)
add_library(stl2stp::core ALIAS stl2stp-core)
target_compile_features(stl2stp-core PUBLIC cxx_std_17)
set_target_properties(stl2stp-core PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
target_include_directories(stl2stp-core PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)

if(STL2STP_WITH_OCCT)
    # OCCT_ROOT is a staged target-platform dependency, outside the Apple SDK.
    # iOS defaults discovery to SDK roots ONLY; do not re-root this explicit
    # prefix. Keep NO_DEFAULT_PATH so unrelated host installs are not selected.
    if(NOT OCCT_INCLUDE_DIR)
        find_path(_occt_headers Standard_Version.hxx
            PATHS "${OCCT_ROOT}/include/opencascade" "${OCCT_ROOT}/inc" "${OCCT_ROOT}/include"
            NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
        if(NOT _occt_headers)
            message(FATAL_ERROR "OCCT headers not found. Set OCCT_ROOT or OCCT_INCLUDE_DIR.")
        endif()
        set(OCCT_INCLUDE_DIR "${_occt_headers}" CACHE PATH "OCCT headers" FORCE)
    endif()
    if(NOT OCCT_LIBRARY_DIR)
        find_library(_occt_kernel NAMES TKernel PATHS "${OCCT_ROOT}/lib" "${OCCT_ROOT}/lib64"
            NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
        if(NOT _occt_kernel)
            message(FATAL_ERROR "OCCT libraries not found. Set OCCT_ROOT or OCCT_LIBRARY_DIR.")
        endif()
        get_filename_component(_occt_library_dir "${_occt_kernel}" DIRECTORY)
        set(OCCT_LIBRARY_DIR "${_occt_library_dir}" CACHE PATH "OCCT libraries" FORCE)
    endif()
    find_package(OpenCASCADE QUIET CONFIG
        PATHS "${OCCT_LIBRARY_DIR}/cmake/opencascade" "${OCCT_LIBRARY_DIR}/cmake/OpenCASCADE"
        NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
    set(_occt_toolkits TKDESTEP TKDEIGES TKXSBase TKBool TKBO TKPrim TKShHealing
        TKTopAlgo TKGeomAlgo TKBRep TKGeomBase TKG3d TKG2d TKMath TKernel)
    set(_occt_links "")
    foreach(_toolkit IN LISTS _occt_toolkits)
        if(TARGET OpenCASCADE::${_toolkit})
            list(APPEND _occt_links OpenCASCADE::${_toolkit})
        elseif(TARGET ${_toolkit})
            list(APPEND _occt_links ${_toolkit})
        else()
            find_library(_occt_${_toolkit} NAMES ${_toolkit} PATHS "${OCCT_LIBRARY_DIR}" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
            if(NOT _occt_${_toolkit})
                message(FATAL_ERROR "Missing OCCT toolkit ${_toolkit} in ${OCCT_LIBRARY_DIR}")
            endif()
            list(APPEND _occt_links "${_occt_${_toolkit}}")
        endif()
    endforeach()
    find_library(_occt_de NAMES TKDE PATHS "${OCCT_LIBRARY_DIR}" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
    if(_occt_de)
        list(APPEND _occt_links "${_occt_de}")
    endif()
    foreach(_direct IN LISTS _occt_links OCCT_EXTRA_LIBRARIES)
        if(_direct MATCHES "TKV3d|TKOpenGl|TKService")
            message(FATAL_ERROR "Core must not directly select visualization toolkit: ${_direct}")
        endif()
    endforeach()
    set(_occt_pending ${_occt_links} ${OCCT_EXTRA_LIBRARIES})
    set(_occt_checked "")
    set(_occt_transitive_visual "")
    while(_occt_pending)
        list(POP_FRONT _occt_pending _dependency)
        if(_dependency MATCHES "TKV3d|TKOpenGl|TKService")
            list(APPEND _occt_transitive_visual "${_dependency}")
        endif()
        if(TARGET "${_dependency}" AND NOT "${_dependency}" IN_LIST _occt_checked)
            list(APPEND _occt_checked "${_dependency}")
            foreach(_property INTERFACE_LINK_LIBRARIES IMPORTED_LINK_INTERFACE_LIBRARIES
                    IMPORTED_LINK_INTERFACE_LIBRARIES_RELEASE IMPORTED_LINK_DEPENDENT_LIBRARIES
                    IMPORTED_LINK_DEPENDENT_LIBRARIES_RELEASE)
                get_target_property(_dependencies "${_dependency}" "${_property}")
                if(_dependencies)
                    foreach(_nested IN LISTS _dependencies)
                        if(_nested MATCHES "^\\$<LINK_ONLY:([^>]+)>$")
                            set(_nested "${CMAKE_MATCH_1}")
                        endif()
                        list(APPEND _occt_pending "${_nested}")
                    endforeach()
                endif()
            endforeach()
        endif()
    endwhile()
    if(_occt_transitive_visual)
        list(REMOVE_DUPLICATES _occt_transitive_visual)
        message(STATUS "OCCT exchange transitive dependencies (not core visualization APIs): ${_occt_transitive_visual}")
    endif()
    find_package(Threads REQUIRED)
    target_sources(stl2stp-core PRIVATE src/exchange.cpp)
    target_include_directories(stl2stp-core SYSTEM PRIVATE "${OCCT_INCLUDE_DIR}")
    target_link_libraries(stl2stp-core PRIVATE ${_occt_links} ${_occt_links}
        ${OCCT_EXTRA_LIBRARIES} Threads::Threads ${CMAKE_DL_LIBS})
    # Tests that call OCCT directly (not only through stl2stp-core) must link these
    # toolkits themselves. Listing them twice plus --no-as-needed is what avoids
    # GNU ld's "DSO missing from command line" on the shared OCCT graph.
    set(STL2STP_OCCT_LIBRARIES ${_occt_links})
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        target_link_options(stl2stp-core PUBLIC "LINKER:--disable-new-dtags")
    endif()
    set(CMAKE_BUILD_RPATH "${OCCT_LIBRARY_DIR}")
    set(CMAKE_INSTALL_RPATH "${OCCT_LIBRARY_DIR}")
    set_target_properties(stl2stp-core PROPERTIES BUILD_RPATH "${OCCT_LIBRARY_DIR}" INSTALL_RPATH "${OCCT_LIBRARY_DIR}")
    message(STATUS "stl2stp OCCT toolkits: ${_occt_toolkits} (TKDE if present)")
else()
    message(WARNING "Mesh-only build: STEP/IGES unavailable; OCCT tests are NOT run.")
    target_sources(stl2stp-core PRIVATE src/exchange_disabled.cpp)
endif()

if(STL2STP_BUILD_CLI)
    add_executable(stl2stp-cli cli/main.cpp)
    target_link_libraries(stl2stp-cli PRIVATE stl2stp-core)
endif()

if(BUILD_TESTING)
    add_subdirectory(tests)
endif()

if(STL2STP_WITH_OCCT)
    install(TARGETS stl2stp-core
        ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
        LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
        RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    if(STL2STP_BUILD_CLI)
        install(TARGETS stl2stp-cli RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    endif()
    install(DIRECTORY include/stl2stp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
endif()
